## 1.auto_ptr

```c++
#include <iostream>
using namespace std;

template<class T>
class autoptr 
{
private:
  T* _ptr;

public:
 // 1.构造函数，缺省的构造函数
  autoptr(T* ptr = nullptr)
    :_ptr(ptr)
  {}

  ~autoptr()
  {
    if(_ptr)
    {
      delete _ptr;
      _ptr = nullptr;
    }
  }

// 3.拷贝构造：移交管理权。把指针的管理权给一个新的对象
   autoptr(autoptr<T>& ap)
     :_ptr(ap._ptr)
   {
      ap._ptr = nullptr;
   }

// 4.赋值构造：移交管理权。
  autoptr<T>& operator=(autoptr<T>& ap)
  {
    if(&ap != this)
    {
      delete _ptr;
      
      // 移交管理权
      _ptr = ap._ptr;
      ap._ptr = nullptr;
    }

    return *this;
  }

  // 1.指针一样解引用
  T& operator*()
  {
    return *_ptr;
  }

  {
    return _ptr;
  }

  // 3.获取原始指针

  {
    return _ptr;
  }
```

## 2.unique_ptr

**`unique_ptr`：独占资源所有权，禁止拷贝，只允许移动。**

```c++
#include <iostream>
using namespace std;

template<class T>
class uniquePtr 
{
private:
  T* _ptr;

public:
// 1.构造函数
  explicit uniquePtr(T* ptr = nullptr)
    :_ptr(ptr)
  {}

// 2.析构函数
  ~uniquePtr()
  {
    delete _ptr;
    _ptr = nullptr;
  }

// 3.进制拷贝构造
  uniquePtr(const uniquePtr<T>&) = delete;

// 4.禁止赋值拷贝
  uniquePtr& operator=(const uniquePtr<T>&) = delete;

// 5.移动构造：转义资源
  uniquePtr(uniquePtr<T>&& up)
    :_ptr(up._ptr)
  {
    up._ptr = nullptr;
  }

// 6.移动赋值：转义资源 
  uniquePtr<T>& operator=(uniquePtr<T>&& up)
  {
    if(&up != this)
    {
      delete _ptr; // 释放自己的资源
      
      _ptr = up._ptr; // 接管对方的资源
      up._ptr = nullptr; // 对方置空
    }

    return *this;
  }

// 使用资源
  // 1.解引用
  T& operator*()
  {
    return * _ptr;
  }

  // 2.指针访问成员
  T* operator->()
  {
    return _ptr;
  }

  // 3.获取原始指针
  T* get() const 
  {
    return _ptr;
  }

  // 4.释放所有权，不delete
  T* release()
  {
    T* tmp = _ptr;
    _ptr = nullptr;
    return tmp;
  }

  // 5.重置资源的管理
  
  void reset(T* ptr = nullptr)
  {
    if(_ptr != nullptr)
    {
      delete _ptr;
      _ptr = ptr;
    }
  }
};

class Date
{
public:
    Date(int year = 2026, int month = 6, int day = 6)
        : _year(year)
        , _month(month)
        , _day(day)
    {}

    void Print()
    {
        cout << _year << "-" << _month << "-" << _day << endl;
    }

private:
    int _year;
    int _month;
    int _day;
};

int main()
{

  uniquePtr<Date> p1(new Date(1, 2, 3));
  p1->Print();

  // 禁止拷贝的和赋值的
  //uniquePtr<Date> p2(p1);
  
  uniquePtr<Date> p2(move(p1));
  p2->Print();

  uniquePtr<Date> p3;
 
  // 移动赋值给新的管理
  p3 = move(p2);

  p3->Print();

// 2. 
  
  // 释放管理权限了，现在由新的指针进行管理
  uniquePtr<int> p(new int{2});
  int* raw = p.release();
  *raw = 999;
  cout<< *raw <<endl;
 // 需要自己管理资源
  
// 释放原来的资源，管理新的资源
  p.reset(raw);

  return 0;
}

```

### 总结

**unique_ptr: 它是独占式智能指针，不能两个对象同时管理同一块资源。**

**核心思想就是：用一个指针指向需要管理的资源。(当unique_ptr需要进行析构的时候，会先析构指向的对象的析构函数， 然后在释放指针的。)**

**移动构造和赋值构造：产生一个新的对象管理资源。**

**release():  释放资源的管理权限，需要自己进行释放的。**

**reset:  先释放旧的资源，然后管理新的资源。**



**c++ delete的时候，会先析构指针指向的资源 ，然后在释放指针指向的空间。**





## 3.shared_ptr

**`shared_ptr` 是共享式智能指针。**
 **多个 `shared_ptr` 可以管理同一块资源，内部通过引用计数记录当前有多少个智能指针正在管理这块资源。**
 **当引用计数变成 `0` 时，才真正释放资源。**

```c++
#include <iostream>
using namespace std;

template<class T>
class SharedPtr
{
private:
    T* _ptr;       // 指向资源
    int* _pCount;  // 指向引用计数
    
public:
    // 构造函数。构造的资源是有效的资源，引用计数==1
    explicit SharedPtr(T* ptr = nullptr)
        : _ptr(ptr)
        , _pCount(nullptr)
    {
        if (_ptr)
        {
            _pCount = new int(1);
        }
    }

    // 拷贝构造。构造的资源是有效的资源，引用及时++。
    SharedPtr(const SharedPtr<T>& sp)
        : _ptr(sp._ptr)
        , _pCount(sp._pCount)
    {
        if (_pCount)
        {
            ++(*_pCount);
        }
    }

    // 赋值重载。先释放原来的资源。（1.原来资源引用计数大于一，减-。 引用计数=1. 直接释放了）
    // 接收新的资源，新的资源是有效资源引用计数++。
    SharedPtr<T>& operator=(const SharedPtr<T>& sp)
    {
        if (this != &sp)
        {
            // 先释放自己原来管理的资源
            Release();

            // 再共享新资源
            _ptr = sp._ptr;
            _pCount = sp._pCount;

            if (_pCount)
            {
                ++(*_pCount);
            }
        }

        return *this;
    }

    // 析构函数
    ~SharedPtr()
    {
        Release();
    }

    // 解引用
    T& operator*()
    {
        return *_ptr;
    }

    // 成员访问
    T* operator->()
    {
        return _ptr;
    }

    // 获取原始指针
    T* get() const
    {
        return _ptr;
    }

    // 获取引用计数
    int use_count() const
    {
        return _pCount ? *_pCount : 0;
    }

private:
    void Release()
    {
        if (_pCount)
        {
            --(*_pCount);

            if (*_pCount == 0)
            {
                delete _ptr;
                delete _pCount;

                _ptr = nullptr;
                _pCount = nullptr;
            }
        }
    }
};
```

### 总结

**多个智能指针共享同一块资源，通过引用计数管理资源生命周期；每拷贝一次，引用计数加一；每析构一次，引用计数减一；当引用计数为 0 时，自动释放资源。**

**核心就是有一个指针 和一个引用计数器。**



```c++
struct Node
{
    std::shared_ptr<Node> next;  // 指向下一个节点
    ~Node() { std::cout << "Node destroyed\n"; }
};

int main() 
{
    auto a = std::make_shared<Node>(); // a的引用计数变成1
    auto b = std::make_shared<Node>(); // b的引用计数变成1
    
    a->next = b;   // a 的 next 指向 b，b 的引用计数 = 2（a 持有 + b 自身）
    b->next = a;   // b 的 next 指向 a，a 的引用计数 = 2（b 持有 + a 自身）
    return 0;      // 离开作用域，a、b 局部变量销毁，各减 1 → 计数变为 1
                   // 但 a 和 b 互相还持有对方，计数永远不到 0，析构函数永远不会被调用
}
```

## 4.weak_ptr

```c++
#include <iostream>
using namespace std;

template<class T>
class WeakPtr;

struct ControlBlock
{
    int _strong;
    int _weak;

    ControlBlock()
        : _strong(1)
        , _weak(0)
    {}
};

template<class T>
class SharedPtr
{
public:
    // 构造函数
    explicit SharedPtr(T* ptr = nullptr)
        : _ptr(ptr)
        , _cb(nullptr)
    {
        if (_ptr)
        {
            _cb = new ControlBlock;
        }
    }

    // 拷贝构造：强引用计数 +1
    SharedPtr(const SharedPtr<T>& sp)
        : _ptr(sp._ptr)
        , _cb(sp._cb)
    {
        if (_cb)
        {
            ++_cb->_strong;
        }
    }

    // 赋值重载
    SharedPtr<T>& operator=(const SharedPtr<T>& sp)
    {
        if (this != &sp)
        {
            Release();

            _ptr = sp._ptr;
            _cb = sp._cb;

            if (_cb)
            {
                ++_cb->_strong;
            }
        }

        return *this;
    }

    // 析构
    ~SharedPtr()
    {
        Release();
    }

    T& operator*()
    {
        return *_ptr;
    }

    T* operator->()
    {
        return _ptr;
    }

    T* get() const
    {
        return _ptr;
    }

    int use_count() const
    {
        return _cb ? _cb->_strong : 0;
    }

private:
    // 给 WeakPtr::lock() 使用的构造函数
    SharedPtr(T* ptr, ControlBlock* cb)
        : _ptr(ptr)
        , _cb(cb)
    {
        if (_cb)
        {
            ++_cb->_strong;
        }
    }

    void Release()
    {
        if (_cb)
        {
            --_cb->_strong;

            // 强引用计数为 0，释放资源
            if (_cb->_strong == 0)
            {
                delete _ptr;
                _ptr = nullptr;

                // 如果弱引用计数也为 0，释放控制块
                if (_cb->_weak == 0)
                {
                    delete _cb;
                    _cb = nullptr;
                }
            }
        }
    }

private:
    T* _ptr;
    ControlBlock* _cb;

    friend class WeakPtr<T>;
};
```

```c++
template<class T>
class WeakPtr
{
public:
    // 默认构造
    WeakPtr()
        : _ptr(nullptr)
        , _cb(nullptr)
    {}

    // 用 SharedPtr 构造 WeakPtr
    WeakPtr(const SharedPtr<T>& sp)
        : _ptr(sp._ptr)
        , _cb(sp._cb)
    {
        if (_cb)
        {
            ++_cb->_weak;
        }
    }

    // 拷贝构造
    WeakPtr(const WeakPtr<T>& wp)
        : _ptr(wp._ptr)
        , _cb(wp._cb)
    {
        if (_cb)
        {
            ++_cb->_weak;
        }
    }

    // 用 SharedPtr 赋值
    WeakPtr<T>& operator=(const SharedPtr<T>& sp)
    {
        Release();

        _ptr = sp._ptr;
        _cb = sp._cb;

        if (_cb)
        {
            ++_cb->_weak;
        }

        return *this;
    }

    // WeakPtr 之间赋值
    WeakPtr<T>& operator=(const WeakPtr<T>& wp)
    {
        if (this != &wp)
        {
            Release();

            _ptr = wp._ptr;
            _cb = wp._cb;

            if (_cb)
            {
                ++_cb->_weak;
            }
        }

        return *this;
    }

    // 析构
    ~WeakPtr()
    {
        Release();
    }

    // 判断资源是否已经释放
    bool expired() const
    {
        return _cb == nullptr || _cb->_strong == 0;
    }

    // 获取强引用计数
    int use_count() const
    {
        return _cb ? _cb->_strong : 0;
    }

    // 尝试提升为 SharedPtr
    SharedPtr<T> lock() const
    {
        if (expired())
        {
            return SharedPtr<T>();
        }

        return SharedPtr<T>(_ptr, _cb);
    }

private:
    void Release()
    {
        if (_cb)
        {
            --_cb->_weak;

            // 如果强引用和弱引用都为 0，释放控制块
            if (_cb->_strong == 0 && _cb->_weak == 0)
            {
                delete _cb;
            }

            _ptr = nullptr;
            _cb = nullptr;
        }
    }

private:
    T* _ptr;
    ControlBlock* _cb;
};
```
